from collections import deque

def safest_path(G, p, s, t):
    # G: 邻接表，G[u]为u的出边列表（每个元素是v）
    # p: 边的事故概率字典，p[(u,v)]为边u→v的p值
    V = len(G)
    # 1. 拓扑排序
    in_degree = {v:0 for v in G}
    for u in G:
        for v in G[u]:
            in_degree[v] += 1
    q = deque()
    topo_order = []
    for v in G:
        if in_degree[v] == 0:
            q.append(v)
    while q:
        u = q.popleft()
        topo_order.append(u)
        for v in G[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                q.append(v)
    
    # 2. 动态规划计算最大安全性
    dp = {v: 0.0 for v in G}
    dp[s] = 1.0  # 起点安全性为1
    for u in topo_order:
        for v in G[u]:
            safe_prob = 1 - p[(u, v)]
            new_safe = dp[u] * safe_prob
            if new_safe > dp[v]:
                dp[v] = new_safe
    return dp[t]

def max_chemical_capacity(c, is_junk):
    # c: 房间容量数组，is_junk: 标记是否为杂物的数组（True=杂物）
    n = len(c)
    if n == 0:
        return 0
    if n == 1:
        return 0 if is_junk[0] else c[0]
    
    # 辅助函数：计算线性序列的最大容量（l到r的房间）
    def linear_max(l, r):
        dp_prev_prev = 0  # dp[i-2]
        dp_prev = 0 if is_junk[l] else c[l]  # dp[i-1]
        for i in range(l+1, r+1):
            current = 0
            if not is_junk[i]:
                current = dp_prev_prev + c[i]
            dp_curr = max(dp_prev, current)
            dp_prev_prev, dp_prev = dp_prev, dp_curr
        return dp_prev
    
    # 情况1：不选第一个房间（r2到rn）
    case1 = linear_max(1, n-1)
    # 情况2：不选最后一个房间（r1到rn-1）
    case2 = linear_max(0, n-2)
    return max(case1, case2)
