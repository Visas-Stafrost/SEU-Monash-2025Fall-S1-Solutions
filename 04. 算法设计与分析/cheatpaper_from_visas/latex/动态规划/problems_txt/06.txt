import heapq

def constrained_shortest_path(G, A, B, s, d, M):
    # G: 邻接表，G[v]为v的邻居列表
    # 状态：(当前路径长度, 当前顶点, 剩余钱数)
    INF = float('inf')
    # 初始化dp：dp[v][m]表示到达v时剩余m元的最短路径长度
    dp = {v: {m: INF for m in range(M+1)} for v in G}
    dp[s][M] = 0
    # 优先队列（小根堆），按路径长度排序
    heap = []
    heapq.heappush(heap, (0, s, M))
    
    while heap:
        dist_u, u, m_u = heapq.heappop(heap)
        # 若已到达终点，直接返回当前距离（优先队列保证是最短）
        if u == d:
            return dist_u
        # 若当前状态不是最优，跳过
        if dist_u > dp[u][m_u]:
            continue
        # 遍历所有邻居v
        for v in G[u]:
            # 经过v需支付B(v)，检查钱数是否足够
            if m_u <= B(v):
                continue
            m_v = m_u - B(v)
            new_dist = dist_u + A[(u, v)]  # A是边长度字典
            # 若新状态更优，更新并加入队列
            if new_dist < dp[v][m_v]:
                dp[v][m_v] = new_dist
                heapq.heappush(heap, (new_dist, v, m_v))
    # 若无可行路径
    return -1
