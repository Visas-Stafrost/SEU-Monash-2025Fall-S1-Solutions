def count_max_subarrays(A):
    n = len(A)
    if n == 0:
        return 0
    # 初始化状态
    current_sum = A[0]
    current_cnt = 1
    max_sum = A[0]
    max_cnt = 1
    
    for i in range(1, n):
        # 更新当前子段和与计数
        if current_sum + A[i] > A[i]:
            current_sum += A[i]
            # 延续之前的子段，计数不变
        elif current_sum + A[i] == A[i]:
            current_sum = A[i]
            current_cnt = 1  # 新子段：仅A[i]
        else:
            current_sum = A[i]
            current_cnt = 1  # 新子段：仅A[i]
        
        # 更新全局最大和与计数
        if current_sum > max_sum:
            max_sum = current_sum
            max_cnt = current_cnt
        elif current_sum == max_sum:
            max_cnt += current_cnt
    return max_cnt

# 测试示例
A = [2, -2, 2]

def min_insertions_to_palindrome(s):
    n = len(s)
    rev_s = s[::-1]
    # 动态规划求LCS长度
    dp = [[0]*(n+1) for _ in range(n+1)]
    for i in range(1, n+1):
        for j in range(1, n+1):
            if s[i-1] == rev_s[j-1]:
                dp[i][j] = dp[i-1][j-1] + 1
            else:
                dp[i][j] = max(dp[i-1][j], dp[i][j-1])
    return n - dp[n][n]

# 测试示例
s = "abcd"
print(min_insertions_to_palindrome(s))  # 输出3

def max_submatrix(A):
    n = len(A)
    if n == 0:
        return 0, (0,0,0,0)
    m = len(A[0])
    # 预处理前缀和数组（行从1开始）
    prefix = [[0]*(m+1) for _ in range(n+1)]
    for j in range(1, m+1):
        for i in range(1, n+1):
            prefix[i][j] = prefix[i-1][j] + A[i-1][j-1]
    
    max_sum = float('-inf')
    best_range = (0,0,0,0)  # (a,b,c,d)：行a-b，列c-d
    
    # 枚举上边界a
    for a in range(1, n+1):
        # 枚举下边界b
        for b in range(a, n+1):
            # 压缩列得到temp数组
            temp = [0]*m
            for j in range(m):
                temp[j] = prefix[b][j+1] - prefix[a-1][j+1]
            # Kadane算法求temp的最大子数组
            current_sum = temp[0]
            current_start = 0
            current_max = temp[0]
            start = 0
            end = 0
            for j in range(1, m):
                if current_sum + temp[j] > temp[j]:
                    current_sum += temp[j]
                else:
                    current_sum = temp[j]
                    current_start = j
                # 更新当前最大子数组
                if current_sum > current_max:
                    current_max = current_sum
                    start = current_start
                    end = j
            # 更新全局最大
            if current_max > max_sum:
                max_sum = current_max
                best_range = (a-1, b-1, start, end)  # 转换为0索引
    return max_sum, best_range

    [1, -2, 3],
    [-4, 5, 1],
    [2, -1, 4]
]
sum_val, (a,b,c,d) = max_submatrix(A)
print(f"最大子矩阵和：{sum_val}")
print(f"行列范围：行{a}~{b}，列{c}~{d}")  # 输出：行1~2，列1~2，和为9
