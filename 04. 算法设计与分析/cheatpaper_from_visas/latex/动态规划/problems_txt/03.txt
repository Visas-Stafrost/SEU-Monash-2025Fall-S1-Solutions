def max_subarray_divide_conquer(A):
    def find_max_crossing_subarray(A, low, mid, high):
        # 计算左半部分最大和（包含mid）
        left_sum = float('-inf')
        current_sum = 0
        for i in range(mid, low-1, -1):
            current_sum += A[i]
            if current_sum > left_sum:
                left_sum = current_sum
        # 计算右半部分最大和（包含mid+1）
        right_sum = float('-inf')
        current_sum = 0
        for i in range(mid+1, high+1):
            current_sum += A[i]
            if current_sum > right_sum:
                right_sum = current_sum
        # 跨越中间的和
        return left_sum + right_sum
    
    def find_max_subarray(A, low, high):
        if low == high:
            return A[low]  # 递归终止：单个元素
        mid = (low + high) // 2
        # 递归求解左、右、跨越中间的最大和
        left_max = find_max_subarray(A, low, mid)
        right_max = find_max_subarray(A, mid+1, high)
        cross_max = find_max_crossing_subarray(A, low, mid, high)
        # 返回三者最大值
        return max(left_max, right_max, cross_max)
    
    return find_max_subarray(A, 0, len(A)-1)

# 测试示例

def longest_non_decreasing_subseq_dp(A):
    n = len(A)
    if n == 0:
        return 0
    dp = [1] * n
    for i in range(1, n):
        for j in range(i):
            if A[j] <= A[i]:
                dp[i] = max(dp[i], dp[j] + 1)
    return max(dp)

import bisect

def longest_non_decreasing_subseq_greedy(A):
    tails = []
    for num in A:
        # 找到第一个大于num的位置（bisect_right对应非降）
        idx = bisect.bisect_right(tails, num)
        if idx == len(tails):
            tails.append(num)
        else:
            tails[idx] = num
    return len(tails)

# 测试示例
